package Sorting;
import java.util.Arrays;

public class MergeDemo {


    // Main merge sort function
    void mergeSort(int[] array, int left, int right) {
        if (left < right) {
            // Find the middle point
            int mid = left + (right - left) / 2;

            // Recursively sort the two halves
            mergeSort(array, left, mid);    // Left half
            mergeSort(array, mid + 1, right); // Right half

            // Merge the sorted halves
            merge(array, left, mid, right);
        }
    }

    // Merge two sorted subarrays into one
    void merge(int[] array, int left, int mid, int right) {
        // Sizes of two subarrays
        int n1 = mid - left + 1;
        int n2 = right - mid;

        // Temporary arrays
        int[] leftArray = new int[n1];
        int[] rightArray = new int[n2];

        // Copy data to temporary arrays
        for (int i = 0; i < n1; i++)
        {
            leftArray[i] = array[left + i];
        }
        for (int j = 0; j < n2; j++){
            rightArray[j] = array[mid + 1 + j];
        }

        // Initial indexes of left and right subarrays
        int a = 0, b = 0;

        // Initial index of merged subarray
        int k = left;

        // Merge the temp arrays back into the original array
        while (a < n1 && b < n2) {
            if (leftArray[a] <= rightArray[b]) {
                array[k] = leftArray[a];
                a++;
            } else {
                array[k] = rightArray[b];
                b++;
            }
            k++;
        }

        // Copy remaining elements of leftArray, if any
        while (a < n1) {
            array[k] = leftArray[a];
            a++;
            k++;
        }

        // Copy remaining elements of rightArray, if any
        while (b < n2) {
            array[k] = rightArray[b];
            b++;
            k++;
        }
    }

    // Driver code
    public static void main(String[] args) {
        int[] array = {12, 11, 13, 5, 6, 7};
        MergeDemo ms = new MergeDemo();

        System.out.println("Original Array: " + Arrays.toString(array));
        ms.mergeSort(array, 0, array.length - 1);
        System.out.println("Sorted Array: " + Arrays.toString(array));
    }
}
