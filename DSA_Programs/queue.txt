package QueueDemo;

public class QueuePrac {
        private int SIZE = 5;  // Maximum size of the queue
        private int[] items = new int[SIZE];  // Array to store queue elements
        private int front = -1;  // Index of the front element
        private int rear = -1;   // Index of the rear element

        // Check if the queue is empty
        boolean isEmpty() {
            return front == -1;
        }

        // Check if the queue is full
        boolean isFull() {
            return rear == SIZE - 1;
        }

        // Add an element to the queue
        void enqueue(int element) {
            if (isFull()) {
                System.out.println("Queue is full! Cannot add " + element);
            } else {
                if (front == -1) {
                    front = 0;  // Set front to 0 if the queue is empty
                }
                items[++rear] = element;  // Add the new element at the rear
                System.out.println("Added: " + element);
            }
        }

        // Remove an element from the queue
        int dequeue() {
            if (isEmpty()) {
                System.out.println("Queue is empty! Cannot dequeue.");
                return -1;  // Return -1 to indicate failure
            } else {
                int element = items[front];  // Get the front element
                if (front > rear) {
                    // Reset the queue if it becomes empty after dequeue
                    front = -1;
                    rear = -1;
                } else {
                    front++;  // Move front to the next position
                }
                System.out.println("Removed: " + element);
                return element;
            }
        }

        // Display the elements of the queue
        void display() {
            if (isEmpty()) {
                System.out.println("Queue is empty!");
            } else {
                System.out.print("Queue elements: ");
                for (int i = front; i <= rear; i++) {
                    System.out.print(items[i] + " ");
                }
                System.out.println();
            }
        }

        // Main method to test the queue
        public static void main(String[] args) {
            QueuePrac queue = new QueuePrac();

            // Enqueue elements
            queue.enqueue(10);
            queue.enqueue(20);
            queue.enqueue(30);
            queue.enqueue(40);
            queue.enqueue(50);

            // Try adding one more element (queue is full)
            queue.enqueue(60);

            // Display queue
            queue.display();

            // Dequeue elements
            queue.dequeue();
            queue.dequeue();

            // Display queue after dequeuing
            queue.display();
        }
    }
